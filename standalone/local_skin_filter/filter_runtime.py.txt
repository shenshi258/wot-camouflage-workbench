# -*- coding: ascii -*-
"""Per-appearance cosmetic resource selection. Never alters collision or visibility.

Loaded by the ordinary gui.mods loader. No memory scanning, injection, network
access, input automation or anti-cheat interaction. Settings affect subsequent
native builds; this module never forces a live battle model rebuild.
"""
import logging
import BigWorld
from vehicle_systems import model_assembler
from vehicle_appearance.common_tank_appearance import CommonTankAppearance
from vehicle_systems.CompoundAppearance import CompoundAppearance

MODEL_PATHS = __MODEL_PATHS__
VEHICLES = __VEHICLES__
_contexts = []
_settings = {}
_installed = False
_log = logging.getLogger('LocalSkinFilter')
LINK = 'local_skin_filter'


def _apply(linkage, values):
    if linkage == LINK and isinstance(values, dict):
        _settings.clear()
        # Only known booleans are accepted. Missing values disable selection.
        for key in ['enabled', 'enemies', 'allies'] + [v['var'] for v in VEHICLES]:
            _settings[key] = values.get(key) is True


def _template():
    help_text = (u'{BODY}\u4ec5\u6062\u590d\u672c\u5730\u66ff\u6362\u6d82\u88c5\uff1b'
                 u'\u4fdd\u7559\u672c\u8f66\u5916\u89c2\u3002\u4fdd\u5b58\u540e\u4e0b\u6b21'
                 u'\u52a0\u8f7d\u8f66\u8f86\u751f\u6548\u3002 / Next vehicle load; own tank kept.{/BODY}')
    rows = [dict(type='CheckBox', text=u'\u654c\u8f66\u4f7f\u7528\u539f\u7248 / Enemies',
                 varName='enemies', value=True, tooltip=help_text),
            dict(type='CheckBox', text=u'\u961f\u53cb\u4f7f\u7528\u539f\u7248 / Allies',
                 varName='allies', value=False, tooltip=help_text)]
    cars = [dict(type='CheckBox', text=v['label'], varName=v['var'], value=True,
                 tooltip=help_text) for v in VEHICLES]
    # The standard API supplies scrolling for longer columns.
    return dict(modDisplayName=u'\u672c\u5730\u6d82\u88c5\u5c4f\u853d / Local skin filter',
                enabled=True, column1=rows, column2=cars)


def _target(vehicle_id):
    """Use only native public team metadata; unknown/observer contexts bypass."""
    try:
        player = BigWorld.player()
        if not _settings.get('enabled') or player is None or vehicle_id is None:
            return False
        own_id = getattr(player, 'playerVehicleID', None)
        if own_id is None or own_id == vehicle_id:
            return False
        # Spectator modes and nonstandard teams have not been validated.
        if player.isObserver():
            return False
        own_team = getattr(player, 'team', None)
        arena = getattr(player, 'arena', None)
        row = getattr(arena, 'vehicles', {}).get(vehicle_id)
        if not isinstance(row, dict):
            return False
        team = row.get('team')
        if own_team not in (1, 2) or team not in (1, 2):
            return False
        return bool(_settings.get('allies' if team == own_team else 'enemies'))
    except Exception:
        return False


def _with_context(original, identify):
    def guarded(appearance, *args, **kwargs):
        _contexts.append(identify(appearance, args, kwargs))
        try:
            return original(appearance, *args, **kwargs)
        finally:
            _contexts.pop()
    return guarded


def init():
    global _installed
    if _installed:
        return
    # Missing settings UI is a deliberate no-op, rather than an invisible MOD.
    from gui.modsSettingsApi import g_modsSettingsApi
    template = _template()
    saved = g_modsSettingsApi.getModSettings(LINK, template)
    if saved:
        g_modsSettingsApi.registerCallback(LINK, _apply)
    else:
        saved = g_modsSettingsApi.setModTemplate(LINK, template, _apply)
    if not isinstance(saved, dict):
        raise RuntimeError('ModsSettings API did not return settings')
    _apply(LINK, saved)
    old_paths = model_assembler.getPartModelsFromDesc
    old_prereqs = CommonTankAppearance.prerequisites
    refresh_name = '_CompoundAppearance__requestModelsRefresh'
    old_refresh = getattr(CompoundAppearance, refresh_name)
    # Pre-compute selection keys; no descriptor, fashion or arena writes.
    vars_by_model = {}
    for source in MODEL_PATHS:
        vehicle = '/'.join(source.split('/')[1:3])
        for spec in VEHICLES:
            if spec['key'] == vehicle:
                vars_by_model[source] = spec['var']
                break

    def part_models(*args, **kwargs):
        models = old_paths(*args, **kwargs)
        if not _contexts or not _contexts[-1]:
            return models
        mapped = []
        for path in models:
            key = path.replace('\\', '/').lower() if isinstance(path, basestring) else None
            mapped.append(MODEL_PATHS.get(key, path)
                          if key in vars_by_model and _settings.get(vars_by_model[key]) else path)
        if list(models) == mapped:
            return models
        if hasattr(models, '_fields'):
            return type(models)(*mapped)
        return tuple(mapped) if isinstance(models, tuple) else mapped

    def prereq_target(appearance, args, kwargs):
        return _target(args[0] if args else kwargs.get('vID'))

    def refresh_target(appearance, args, kwargs):
        return _target(getattr(appearance, '_CommonTankAppearance__vID', None))

    model_assembler.getPartModelsFromDesc = part_models
    CommonTankAppearance.prerequisites = _with_context(old_prereqs, prereq_target)
    setattr(CompoundAppearance, refresh_name, _with_context(old_refresh, refresh_target))
    _installed = True
    _log.info('Cosmetic filter ready; %d vehicles. No in-game validation claimed.', len(VEHICLES))


try:
    init()
except Exception:
    _log.exception('Cosmetic filter unavailable; native selection retained')
